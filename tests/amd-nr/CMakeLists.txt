cmake_minimum_required(VERSION 3.21)
project(TRPAmdNrFoundations LANGUAGES CXX)
enable_testing()
include(../../cmake/AmdNrFoundations.cmake)
add_executable(TRPAmdNrArchiveTests WeightArchiveTests.cpp)
target_compile_features(TRPAmdNrArchiveTests PRIVATE cxx_std_23)
target_include_directories(TRPAmdNrArchiveTests PRIVATE ../../src)
target_link_libraries(TRPAmdNrArchiveTests PRIVATE TRPAmdNrCore)
add_test(NAME AmdNrArchive COMMAND TRPAmdNrArchiveTests)
add_executable(TRPAmdNrProjectionTests C512ProjectionTests.cpp)
target_link_libraries(TRPAmdNrProjectionTests PRIVATE TRPAmdNrCore)
add_test(NAME AmdNrC512Projection COMMAND TRPAmdNrProjectionTests)
add_executable(TRPAmdNrTensorTests C512TensorTests.cpp)
target_link_libraries(TRPAmdNrTensorTests PRIVATE TRPAmdNrCore)
add_test(NAME AmdNrC512Tensor COMMAND TRPAmdNrTensorTests)
add_executable(TRPAmdNrProjectionArchiveTests C512ProjectionArchiveTests.cpp)
target_link_libraries(TRPAmdNrProjectionArchiveTests PRIVATE TRPAmdNrCore)
add_test(NAME AmdNrC512ProjectionArchive COMMAND TRPAmdNrProjectionArchiveTests)
add_executable(TRPAmdNrNumericTests NumericFormatsTests.cpp)
target_link_libraries(TRPAmdNrNumericTests PRIVATE TRPAmdNrCore)
add_test(NAME AmdNrNumericFormats COMMAND TRPAmdNrNumericTests)
add_executable(TRPAmdNrGeometryTests ModelGeometryTests.cpp)
target_link_libraries(TRPAmdNrGeometryTests PRIVATE TRPAmdNrCore)
add_test(NAME AmdNrGeometry COMMAND TRPAmdNrGeometryTests)
add_executable(TRPAmdNrIdentityTests ModelIdentityTests.cpp)
target_link_libraries(TRPAmdNrIdentityTests PRIVATE TRPAmdNrCore)
add_test(NAME AmdNrIdentity COMMAND TRPAmdNrIdentityTests)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME AmdNrInspector COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/InspectorTests.py $<TARGET_FILE:TRPAmdNrInspect>)
add_executable(TRPAmdNrGpuTests FormatProbeTests.cpp)
target_link_libraries(TRPAmdNrGpuTests PRIVATE TRPAmdNrProbe)
add_test(NAME AmdNrGpuFormats COMMAND TRPAmdNrGpuTests)
set_tests_properties(AmdNrGpuFormats PROPERTIES SKIP_RETURN_CODE 77)
add_test(NAME AmdNrGpuStorage COMMAND TRPAmdNrGpuTests --without-debug)
add_test(NAME AmdNrGpuCli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/FormatProbeCliTests.py $<TARGET_FILE:TRPAmdNrFormatProbe>)
execute_process(COMMAND ${Python3_EXECUTABLE} -c "import numpy" RESULT_VARIABLE AMD_NR_NUMPY_STATUS OUTPUT_QUIET ERROR_QUIET)
if(AMD_NR_NUMPY_STATUS EQUAL 0)
    add_test(NAME AmdNrOracleSelfTest COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/NumericOracle.py --self-test)
    add_test(NAME AmdNrOracleCpu COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/NumericOracle.py --cpu $<TARGET_FILE:TRPAmdNrInspect> --work-dir ${CMAKE_CURRENT_BINARY_DIR}/oracle-cpu)
    add_test(NAME AmdNrOracleWarp COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/NumericOracle.py --cpu $<TARGET_FILE:TRPAmdNrInspect> --gpu $<TARGET_FILE:TRPAmdNrFormatProbe> --warp --work-dir ${CMAKE_CURRENT_BINARY_DIR}/oracle-warp)
else()
    message(WARNING "NumPy unavailable: independent AMD NR numerical oracle tests are not registered")
endif()
