# Compile the production lifecycle adapter against CPU ownership facades. The
# fixture cannot execute GPU work or load a vendor runtime.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${PROJECT_SOURCE_DIR}/src/FrameGen/NvidiaHostLifecycle.cpp")
file(READ "${PROJECT_SOURCE_DIR}/src/FrameGen/NvidiaHostLifecycle.cpp" lifecycleSource)
string(FIND "${lifecycleSource}" "struct NvidiaHost::LifecycleOperations" adapterBegin)
string(FIND "${lifecycleSource}" "HRESULT NvidiaHost::FailLifecycle" adapterEnd)
math(EXPR adapterLength "${adapterEnd}-${adapterBegin}")
string(SUBSTRING "${lifecycleSource}" ${adapterBegin} ${adapterLength} lifecycleAdapter)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/LifecycleAdapter.inc" "${lifecycleAdapter}")
string(FIND "${lifecycleSource}" "HRESULT NvidiaHost::BeforeResizeBuffers" resizeBegin)
string(FIND "${lifecycleSource}" "HRESULT NvidiaHost::AfterResizeBuffers" resizeEnd)
math(EXPR resizeLength "${resizeEnd}-${resizeBegin}")
string(SUBSTRING "${lifecycleSource}" ${resizeBegin} ${resizeLength} beforeResize)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/BeforeResize.inc" "${beforeResize}")
add_executable(TRPFsrHostResizeTests FsrHostResizeTests.cpp)
target_compile_features(TRPFsrHostResizeTests PRIVATE cxx_std_23)
target_compile_definitions(TRPFsrHostResizeTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN TRP_ENABLE_FSR)
target_include_directories(TRPFsrHostResizeTests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${PROJECT_SOURCE_DIR}/src")
add_test(NAME FsrHostResizeOwnership COMMAND TRPFsrHostResizeTests)
if(TRP_ENABLE_XESS)
    add_executable(TRPXessHostResizeTests FsrHostResizeTests.cpp)
    target_compile_features(TRPXessHostResizeTests PRIVATE cxx_std_23)
    target_compile_definitions(TRPXessHostResizeTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN TRP_ENABLE_FSR TRP_ENABLE_XESS)
    target_include_directories(TRPXessHostResizeTests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${PROJECT_SOURCE_DIR}/src")
    add_test(NAME XessHostResizeAffinity COMMAND TRPXessHostResizeTests)
endif()

set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/src/FrameGen/NvidiaHost.cpp" "${PROJECT_SOURCE_DIR}/src/FrameGen/NvidiaHostStartup.cpp")
file(READ "${PROJECT_SOURCE_DIR}/src/FrameGen/NvidiaHost.cpp" mixedSource)
string(FIND "${mixedSource}" "bool NvidiaHost::EvaluateFrame" mixedBegin)
string(FIND "${mixedSource}" "float NvidiaHost::OptimalMipmapBias" mixedEnd)
math(EXPR mixedLength "${mixedEnd}-${mixedBegin}")
string(SUBSTRING "${mixedSource}" ${mixedBegin} ${mixedLength} mixedEvaluate)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/MixedEvaluate.inc" "${mixedEvaluate}")
file(READ "${PROJECT_SOURCE_DIR}/src/FrameGen/NvidiaHostStartup.cpp" startupSource)
string(FIND "${startupSource}" "HRESULT NvidiaHost::ResizeFsrSwapChain" mixedResizeBegin)
string(SUBSTRING "${startupSource}" ${mixedResizeBegin} -1 mixedResize)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/MixedResize.inc" "${mixedResize}")
add_executable(TRPMixedHostTests MixedHostTests.cpp)
target_compile_features(TRPMixedHostTests PRIVATE cxx_std_23)
target_compile_definitions(TRPMixedHostTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN TRP_ENABLE_FSR TRP_ENABLE_FSR_FG)
target_include_directories(TRPMixedHostTests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${PROJECT_SOURCE_DIR}/src")
add_test(NAME FsrMixedHostRecovery COMMAND TRPMixedHostTests)
