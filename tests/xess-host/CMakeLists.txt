# Like the existing FSR host fixture, compile the actual host operation bodies
# without Skyrim dependencies. SDK/bridge execution has separate real GPU tests.
set(source "${PROJECT_SOURCE_DIR}/src/FrameGen/SourceXessEvaluation.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
file(READ "${source}" sourceText)
set(bodies "")
foreach(method EvaluateOptionalPreUpscale EvaluateOptionalPostUpscale EvaluateUpscaler PrepareGeneration)
    string(FIND "${sourceText}" "    bool ${method}(" begin)
    if(method STREQUAL EvaluateUpscaler)
        string(FIND "${sourceText}" "    Result<UpscaleOutcome> ${method}(" begin)
    endif()
    if(method STREQUAL PrepareGeneration)
        string(FIND "${sourceText}" "    GenerationPreparationStatus ${method}(" begin)
    endif()
    string(SUBSTRING "${sourceText}" ${begin} -1 tail)
    string(FIND "${tail}" "\n    }" end)
    math(EXPR length "${end}+6")
    string(SUBSTRING "${tail}" 0 ${length} body)
    string(REPLACE "${method}(" "NvidiaHost::SourceXessEvaluationOperations::${method}(" body "${body}")
    string(APPEND bodies "${body}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/XessHostOperations.inc" "${bodies}")
add_executable(TRPXessHostFrameTests XessHostFrameTests.cpp)
target_link_libraries(TRPXessHostFrameTests PRIVATE TRPXessRuntime)
target_include_directories(TRPXessHostFrameTests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME XessHostFrameRecovery COMMAND TRPXessHostFrameTests)
