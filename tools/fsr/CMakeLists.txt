cmake_minimum_required(VERSION 3.21)
project(TRPFsrProviderProbe LANGUAGES CXX)
get_filename_component(TRP_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(TRP_FSR_SDK_DIR "" CACHE PATH "Verified pinned SDK root")
file(READ "${TRP_ROOT}/tools/fsr/runtime-pin.json" pin)
foreach(index RANGE 0 3)
    string(JSON headerPath GET "${pin}" headers ${index} path)
    string(JSON expectedHash GET "${pin}" headers ${index} sha256)
    file(SHA256 "${TRP_FSR_SDK_DIR}/${headerPath}" actualHash)
    if(NOT actualHash STREQUAL expectedHash)
        message(FATAL_ERROR "FidelityFX header SHA-256 differs: ${headerPath}")
    endif()
endforeach()
add_executable(TRPFsrProviderProbe Probe-Providers.cpp "${TRP_ROOT}/src/Upscaling/FSRRuntime.cpp")
target_compile_features(TRPFsrProviderProbe PRIVATE cxx_std_23)
target_compile_definitions(TRPFsrProviderProbe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(TRPFsrProviderProbe PRIVATE "${TRP_ROOT}/src" "${TRP_FSR_SDK_DIR}/Kits/FidelityFX/api/include" "${TRP_FSR_SDK_DIR}/Kits/FidelityFX/upscalers/include")
target_link_libraries(TRPFsrProviderProbe PRIVATE d3d11 d3d12 dxgi)
