cmake_minimum_required(VERSION 3.25)
project(NrFormatResearch LANGUAGES CXX)
enable_testing()
get_filename_component(nrRoot "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_subdirectory("${nrRoot}/tests/nr-runtime" nr-runtime)
if(NOT TARGET TRPNrPerformanceRuntime)
    message(FATAL_ERROR "Configure TRP_NR_NGX_INCLUDE_DIR with the pinned local NGX SDK headers")
endif()
# Use the production typed SDR packet admission; historical experiments used
# private format overrides whose original receipts remain unchanged.
add_executable(NrFormatContextProbe FormatContextProbe.cpp)
target_compile_features(NrFormatContextProbe PRIVATE cxx_std_23)
target_compile_definitions(NrFormatContextProbe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(NrFormatContextProbe PRIVATE "${nrRoot}/tests")
target_link_libraries(NrFormatContextProbe PRIVATE TRPNrPerformanceRuntime)
