cmake_minimum_required(VERSION 3.25)
project(NrFormatResearch LANGUAGES CXX)
enable_testing()
get_filename_component(nrRoot "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_subdirectory("${nrRoot}/tests/nr-runtime" nr-runtime)
if(NOT TARGET TRPNrPerformanceRuntime)
    message(FATAL_ERROR "Configure TRP_NR_NGX_INCLUDE_DIR with the pinned local NGX SDK headers")
endif()
file(READ "${nrRoot}/src/NeuralRendering/Stage.cpp" stageSource)
file(READ "${nrRoot}/src/NeuralRendering/ImagePacket.cpp" packetSource)
set(stageAnchor "srv.Format=DXGI_FORMAT_R16G16B16A16_FLOAT;")
set(packetAnchor "d.Format==format&&")
string(FIND "${stageSource}" "${stageAnchor}" stageIndex)
string(FIND "${packetSource}" "${packetAnchor}" packetIndex)
if(stageIndex LESS 0 OR packetIndex LESS 0)
    message(FATAL_ERROR "Research source anchors changed; review format experiment before rebuilding")
endif()
string(REPLACE "${stageAnchor}" "srv.Format=packet.color->GetDesc().Format;" stageSource "${stageSource}")
string(REPLACE "${packetAnchor}" "(d.Format==format||(format==DXGI_FORMAT_R16G16B16A16_FLOAT&&d.Format==DXGI_FORMAT_R8G8B8A8_UNORM))&&" packetSource "${packetSource}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ResearchFormatStage.cpp" "${stageSource}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ResearchFormatPacket.cpp" "${packetSource}")
get_target_property(runtimeSources TRPNrPerformanceRuntime SOURCES)
list(REMOVE_ITEM runtimeSources "${nrRoot}/src/NeuralRendering/Stage.cpp" "${nrRoot}/src/NeuralRendering/ImagePacket.cpp")
list(APPEND runtimeSources "${CMAKE_CURRENT_BINARY_DIR}/ResearchFormatStage.cpp" "${CMAKE_CURRENT_BINARY_DIR}/ResearchFormatPacket.cpp")
set_property(TARGET TRPNrPerformanceRuntime PROPERTY SOURCES "${runtimeSources}")
target_include_directories(TRPNrPerformanceRuntime PRIVATE "${nrRoot}/src/NeuralRendering")
add_executable(NrFormatContextProbe FormatContextProbe.cpp)
target_compile_features(NrFormatContextProbe PRIVATE cxx_std_23)
target_compile_definitions(NrFormatContextProbe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(NrFormatContextProbe PRIVATE "${nrRoot}/tests")
target_link_libraries(NrFormatContextProbe PRIVATE TRPNrPerformanceRuntime)
