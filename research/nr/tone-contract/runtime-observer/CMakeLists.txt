cmake_minimum_required(VERSION 3.25)
project(NrBoundaryResearch LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The pinned observer requires Windows x64")
endif()
enable_testing()
add_executable(NrObserverFixture Fixture.cpp)
add_executable(NrBoundaryObserver Observer.cpp)
foreach(target NrObserverFixture NrBoundaryObserver)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(${target} PRIVATE /W4 /WX)
endforeach()
target_link_libraries(NrBoundaryObserver PRIVATE bcrypt psapi)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME observer-lifecycle COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/TestObserver.py" --bin "$<TARGET_FILE_DIR:NrBoundaryObserver>" -v)
set_tests_properties(observer-lifecycle PROPERTIES TIMEOUT 60)
