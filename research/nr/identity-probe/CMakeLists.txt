# Build private source copies. Never change production ownership or ABI.
function(replace_required variable anchor replacement)
    set(content "${${variable}}")
    string(FIND "${content}" "${anchor}" position)
    if(position LESS 0)
        message(FATAL_ERROR "Identity experiment anchor changed: ${anchor}")
    endif()
    string(LENGTH "${anchor}" length)
    math(EXPR after "${position}+${length}")
    string(SUBSTRING "${content}" ${after} -1 tail)
    string(FIND "${tail}" "${anchor}" duplicate)
    if(NOT duplicate LESS 0)
        message(FATAL_ERROR "Identity experiment anchor is ambiguous: ${anchor}")
    endif()
    string(REPLACE "${anchor}" "${replacement}" content "${content}")
    set(${variable} "${content}" PARENT_SCOPE)
endfunction()
