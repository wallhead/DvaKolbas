cmake_minimum_required(VERSION 3.25)
project(NrMotionResearch LANGUAGES CXX)
get_filename_component(nrRoot "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${nrRoot}/cmake/NeuralRuntime.cmake")
set(NR_NGX_INCLUDE "" CACHE PATH "Local pinned NGX SDK headers")
if(NOT EXISTS "${NR_NGX_INCLUDE}/nvsdk_ngx_params.h")
    message(FATAL_ERROR "Supply NR_NGX_INCLUDE with the existing local SDK")
endif()
trp_nr_enable_source_stage("${NR_NGX_INCLUDE}")
# Controlled RE parity experiment, never a product option or source edit.
# The recovered AIO immediate route falls back to Output for Backbuffer.
option(NR_RE_BACKBUFFER_OUTPUT "Research only: bind Backbuffer to Output" OFF)
option(NR_RE_AUTO_EXPOSURE_OFF "Research only: clear NGX AutoExposure creation flag" OFF)
if(NR_RE_BACKBUFFER_OUTPUT AND NR_RE_AUTO_EXPOSURE_OFF)
    message(FATAL_ERROR "Compare one research parameter at a time")
endif()
if(NR_RE_BACKBUFFER_OUTPUT OR NR_RE_AUTO_EXPOSURE_OFF)
    file(READ "${nrRoot}/src/NeuralRendering/Stage.cpp" stageSource)
    if(NR_RE_BACKBUFFER_OUTPUT)
        set(anchor "p.Set(\"DLSSNR.UI\",static_cast<ID3D12Resource*>(nullptr));")
        string(FIND "${stageSource}" "${anchor}" anchorIndex)
        if(anchorIndex LESS 0)
            message(FATAL_ERROR "Research Stage anchor changed; inspect before using variant")
        endif()
        string(REPLACE "${anchor}" "p.Set(\"DLSSNR.Backbuffer\",packet.output.Get());\n    p.Set(\"DLSSNR.BackbufferSubrectBaseX\",0u);p.Set(\"DLSSNR.BackbufferSubrectBaseY\",0u);\n    p.Set(\"DLSSNR.BackbufferSubrectWidth\",packet.colorExtent.width);p.Set(\"DLSSNR.BackbufferSubrectHeight\",packet.colorExtent.height);\n    ${anchor}" stageSource "${stageSource}")
    endif()
    if(NR_RE_AUTO_EXPOSURE_OFF)
        file(READ "${nrRoot}/src/NeuralRendering/RuntimeParameters.h" parameterSource)
        set(flagAnchor "p.Set(\"DLSS.Feature.Create.Flags\",0x42);")
        string(FIND "${parameterSource}" "${flagAnchor}" flagIndex)
        if(flagIndex LESS 0)
            message(FATAL_ERROR "NR creation-flag anchor changed; inspect before using variant")
        endif()
        string(REPLACE "${flagAnchor}" "p.Set(\"DLSS.Feature.Create.Flags\",0x02);" parameterSource "${parameterSource}")
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ResearchRuntimeParameters.h" "${parameterSource}")
        string(REPLACE "\"RuntimeParameters.h\"" "\"ResearchRuntimeParameters.h\"" stageSource "${stageSource}")
    endif()
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ResearchStage.cpp" "${stageSource}")
    get_target_property(runtimeSources TRPNeuralRuntime SOURCES)
    list(REMOVE_ITEM runtimeSources "${nrRoot}/src/NeuralRendering/Stage.cpp")
    list(APPEND runtimeSources "${CMAKE_CURRENT_BINARY_DIR}/ResearchStage.cpp")
    set_property(TARGET TRPNeuralRuntime PROPERTY SOURCES "${runtimeSources}")
    target_include_directories(TRPNeuralRuntime PRIVATE "${nrRoot}/src/NeuralRendering")
endif()
add_executable(NrMotionProbe MotionProbe.cpp)
target_compile_features(NrMotionProbe PRIVATE cxx_std_23)
target_compile_definitions(NrMotionProbe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(NrMotionProbe PRIVATE "${nrRoot}/tests")
target_link_libraries(NrMotionProbe PRIVATE TRPNeuralRuntime)
